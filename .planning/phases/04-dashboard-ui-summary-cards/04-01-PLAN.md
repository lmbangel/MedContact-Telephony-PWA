Create 4 query sets (each with 5 time filter variants: today/yesterday/this_week/this_month/custom):

**1. GetCallStatsByAgentForCompany** - Admin sees all agents in their company
```sql
-- name: GetCallStatsByAgentForCompanyToday :many
SELECT
  u.id as agent_id,
  CONCAT(u.first_name, ' ', u.last_name) as agent_name,
  COUNT(c.id) as total_calls,
  SUM(CASE WHEN c.status = 'completed' THEN 1 ELSE 0 END) as answered_calls,
  SUM(CASE WHEN c.status = 'missed' OR c.status = 'no-answer' THEN 1 ELSE 0 END) as missed_calls,
  COALESCE(AVG(c.duration), 0) as avg_duration
FROM users u
LEFT JOIN calls c ON u.id = c.user_id
  AND DATE(c.created_at) = CURDATE()
WHERE u.company_id = ?
  AND u.is_active = 1
  AND u.role IN ('agent', 'manager', 'supervisor')
GROUP BY u.id, u.first_name, u.last_name
ORDER BY total_calls DESC;
```

**2. GetTaskStatsByAgentForCompany** - Task metrics per agent for admin
```sql
-- name: GetTaskStatsByAgentForCompanyToday :many
SELECT
  u.id as agent_id,
  CONCAT(u.first_name, ' ', u.last_name) as agent_name,
  COUNT(t.id) as total_tasks,
  SUM(CASE WHEN t.status = 'completed' THEN 1 ELSE 0 END) as completed_tasks,
  SUM(CASE WHEN t.status = 'pending' THEN 1 ELSE 0 END) as pending_tasks
FROM users u
LEFT JOIN tasks t ON u.id = t.assigned_to
  AND DATE(t.created_at) = CURDATE()
WHERE u.company_id = ?
  AND u.is_active = 1
  AND u.role IN ('agent', 'manager', 'supervisor')
GROUP BY u.id, u.first_name, u.last_name
ORDER BY total_tasks DESC;
```

**3. GetCallStatsByAgentForManager** - Manager sees their direct/indirect reports (with recursive CTE)
```sql
-- name: GetCallStatsByAgentForManagerToday :many
WITH RECURSIVE team AS (
  SELECT id FROM users WHERE id = ?
  UNION
  SELECT u.id FROM users u
  INNER JOIN team t ON u.reports_to = t.id
  WHERE u.company_id = ? AND u.is_active = 1
)
SELECT
  u.id as agent_id,
  CONCAT(u.first_name, ' ', u.last_name) as agent_name,
  COUNT(c.id) as total_calls,
  SUM(CASE WHEN c.status = 'completed' THEN 1 ELSE 0 END) as answered_calls,
  SUM(CASE WHEN c.status = 'missed' OR c.status = 'no-answer' THEN 1 ELSE 0 END) as missed_calls,
  COALESCE(AVG(c.duration), 0) as avg_duration
FROM users u
INNER JOIN team ON u.id = team.id
LEFT JOIN calls c ON u.id = c.user_id
  AND DATE(c.created_at) = CURDATE()
WHERE u.id != ?
GROUP BY u.id, u.first_name, u.last_name
ORDER BY total_calls DESC;
```

**Helper function: getAgentBreakdownForCompany**
```go
func (h *StatsHandler) getAgentBreakdownForCompany(r *http.Request, companyID string, filterType string) ([]AgentBreakdown, error) {
    // Fetch call stats by agent
    var callStats []db.GetCallStatsByAgentForCompanyTodayRow
    switch filterType {
    case "today":
        callStats, err = h.queries.GetCallStatsByAgentForCompanyToday(r.Context(), companyID)
    case "yesterday":
        callStats, err = h.queries.GetCallStatsByAgentForCompanyYesterday(r.Context(), companyID)
    // ... other filters
    }

    // Fetch task stats by agent
    var taskStats []db.GetTaskStatsByAgentForCompanyTodayRow
    switch filterType {
    case "today":
        taskStats, err = h.queries.GetTaskStatsByAgentForCompanyToday(r.Context(), companyID)
    // ... other filters
    }

    // Merge call and task stats by agent_id
    agentMap := make(map[string]*AgentBreakdown)
    for _, c := range callStats {
        agentMap[c.AgentID] = &AgentBreakdown{
            AgentID: c.AgentID,
            AgentName: c.AgentName,
            TotalCalls: c.TotalCalls,
            AnsweredCalls: c.AnsweredCalls,
            MissedCalls: c.MissedCalls,
            AvgDuration: c.AvgDuration,
        }
    }
    for _, t := range taskStats {
        if agent, ok := agentMap[t.AgentID]; ok {
            agent.TotalTasks = t.TotalTasks
            agent.CompletedTasks = t.CompletedTasks
            agent.PendingTasks = t.PendingTasks
        } else {
            agentMap[t.AgentID] = &AgentBreakdown{
                AgentID: t.AgentID,
                AgentName: t.AgentName,
                TotalTasks: t.TotalTasks,
                CompletedTasks: t.CompletedTasks,
                PendingTasks: t.PendingTasks,
            }
        }
    }

    // Convert map to slice
    agents := make([]AgentBreakdown, 0, len(agentMap))
    for _, a := range agentMap {
        agents = append(agents, *a)
    }
    return agents, nil
}
```
