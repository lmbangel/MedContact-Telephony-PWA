```javascript
/**
 * Initialize charts with memory-safe patterns
 * - Single instance creation (never recreate)
 * - Bounded data windowing (max 100 points)
 * - ResizeObserver for responsive rendering
 */
function initializeCharts() {
  // Call Volume Line Chart
  const volumeContainer = document.getElementById('callVolumeChart');
  if (volumeContainer && !callVolumeChart) {
    const canvas = document.createElement('canvas');
    volumeContainer.appendChild(canvas);
    const ctx = canvas.getContext('2d');

    callVolumeChart = new Chart(ctx, {
      type: 'line',
      data: {
        labels: [], // Time labels (populated by API)
        datasets: [{
          label: 'Call Volume',
          data: [], // Call counts (populated by API)
          borderColor: '#2563eb',
          backgroundColor: 'rgba(37, 99, 235, 0.05)',
          tension: 0.1,
          fill: true
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        animation: { duration: 300 },
        plugins: {
          legend: { display: true, position: 'top' },
          tooltip: { enabled: true, mode: 'index', intersect: false }
        },
        scales: {
          y: {
            beginAtZero: true,
            ticks: {
              callback: function(value) {
                return value.toLocaleString();
              }
            }
          }
        }
      }
    });

    // Setup responsive resize with ResizeObserver
    setupChartResize(callVolumeChart, volumeContainer);
  }

  // Agent Performance Bar Chart
  const perfContainer = document.getElementById('agentPerformanceChart');
  if (perfContainer && !agentPerformanceChart) {
    const canvas = document.createElement('canvas');
    perfContainer.appendChild(canvas);
    const ctx = canvas.getContext('2d');

    agentPerformanceChart = new Chart(ctx, {
      type: 'bar',
      data: {
        labels: [], // Agent names (populated by API)
        datasets: [
          {
            label: 'Total Calls',
            data: [], // Call counts per agent
            backgroundColor: '#2563eb'
          },
          {
            label: 'Answered Calls',
            data: [], // Answered counts per agent
            backgroundColor: '#10b981'
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        animation: { duration: 300 },
        plugins: {
          legend: { display: true, position: 'top' },
          tooltip: { enabled: true, mode: 'index', intersect: false }
        },
        scales: {
          y: {
            beginAtZero: true,
            ticks: {
              callback: function(value) {
                return value.toLocaleString();
              }
            }
          }
        }
      }
    });

    // Setup responsive resize
    setupChartResize(agentPerformanceChart, perfContainer);
  }
}

/**
 * Setup ResizeObserver for responsive chart rendering
 * Throttles with requestAnimationFrame to prevent excessive redraws
 */
let resizeThrottleId = null;
function setupChartResize(chart, container) {
  const resizeObserver = new ResizeObserver(() => {
    if (resizeThrottleId) {
      cancelAnimationFrame(resizeThrottleId);
    }
    resizeThrottleId = requestAnimationFrame(() => {
      chart.resize();
    });
  });
  resizeObserver.observe(container);
}

/**
 * Cleanup charts on page unload to prevent memory leaks
 */
function cleanupCharts() {
  if (callVolumeChart) {
    callVolumeChart.destroy();
    callVolumeChart = null;
  }
  if (agentPerformanceChart) {
    agentPerformanceChart.destroy();
    agentPerformanceChart = null;
  }
  if (resizeThrottleId) {
    cancelAnimationFrame(resizeThrottleId);
    resizeThrottleId = null;
  }
}
```

Why these patterns (from research):
- Single instance creation prevents memory leaks from recreation
- Bounded windowing (MAX_CHART_DATA_POINTS) prevents unbounded growth
- ResizeObserver + requestAnimationFrame throttling prevents resize jank
- chart.destroy() on beforeunload releases memory properly
- Canvas rendering (Chart.js) is faster than SVG for real-time updates
- maintainAspectRatio: false lets Tailwind h-80 control height

```javascript
function loadPlaceholderChartData() {
  // Line chart: last 10 time periods with sample call volumes
  if (callVolumeChart) {
    const now = new Date();
    const labels = [];
    const data = [];

    for (let i = 9; i >= 0; i--) {
      const time = new Date(now - i * 15 * 60 * 1000); // 15-minute intervals
      labels.push(time.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }));
      data.push(Math.floor(Math.random() * 20) + 5); // Random 5-25 calls
    }

    callVolumeChart.data.labels = labels;
    callVolumeChart.data.datasets[0].data = data;
    callVolumeChart.update('none');
  }

  // Bar chart: sample agent performance
  if (agentPerformanceChart) {
    const agents = ['Alice Smith', 'Bob Johnson', 'Carol Williams', 'David Brown'];
    const totalCalls = [42, 38, 51, 29];
    const answeredCalls = [39, 35, 48, 26];

    agentPerformanceChart.data.labels = agents;
    agentPerformanceChart.data.datasets[0].data = totalCalls;
    agentPerformanceChart.data.datasets[1].data = answeredCalls;
    agentPerformanceChart.update('none');
  }
}
```
