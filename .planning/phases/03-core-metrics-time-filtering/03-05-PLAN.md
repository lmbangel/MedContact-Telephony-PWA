INSERT INTO transcriptions (company_id, agent_id, phone_number, recording_url, created_at)
SELECT
  1 as company_id,
  1 as agent_id,
  CONCAT('+1555', LPAD(FLOOR(RAND() * 10000000), 7, '0')) as phone_number,
  CONCAT('https://recordings.example.com/', UUID()) as recording_url,
  DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30) DAY) + INTERVAL FLOOR(RAND() * 86400) SECOND as created_at

-- Insert 10,000 tasks over past 30 days
INSERT INTO tasks (assigned_to, company_id, title, description, status, priority, due_date, created_at)
SELECT
  1 as assigned_to,
  1 as company_id,
  CONCAT('Test Task ', FLOOR(RAND() * 100000)) as title,
  'Performance test data' as description,
  ELT(FLOOR(1 + RAND() * 5), 'pending', 'in-progress', 'completed', 'cancelled', 'follow-up') as status,
  ELT(FLOOR(1 + RAND() * 3), 'low', 'medium', 'high') as priority,
  DATE_ADD(NOW(), INTERVAL FLOOR(RAND() * 7) DAY) as due_date,
  DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30) DAY) + INTERVAL FLOOR(RAND() * 86400) SECOND as created_at
FROM
  (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) t1,
  (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) t2,
  (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) t3,

Run EXPLAIN on key queries to verify index usage, then measure API response times.

**Query plan analysis (save to performance_test_results.txt):**

```bash
cat > run_performance_tests.sh << 'EOF'
#!/bin/bash

echo "=== Performance Test Results ===" > performance_test_results.txt
echo "Generated: $(date)" >> performance_test_results.txt
echo "" >> performance_test_results.txt

# Check data volume
echo "--- Data Volume ---" >> performance_test_results.txt
mysql -u root -proot omnicall -e "SELECT COUNT(*) as transcriptions_count FROM transcriptions" >> performance_test_results.txt
mysql -u root -proot omnicall -e "SELECT COUNT(*) as tasks_count FROM tasks" >> performance_test_results.txt
echo "" >> performance_test_results.txt

# EXPLAIN call stats queries
echo "--- Query Plan: GetCallStatsByCompanyToday ---" >> performance_test_results.txt
mysql -u root -proot omnicall -e "EXPLAIN SELECT COUNT(*) as total_calls, SUM(CASE WHEN answered = 1 THEN 1 ELSE 0 END) as answered_calls, SUM(CASE WHEN answered = 0 THEN 1 ELSE 0 END) as missed_calls, AVG(CASE WHEN duration > 0 THEN duration ELSE NULL END) as avg_duration FROM transcriptions WHERE company_id = 1 AND DATE(created_at) = CURDATE()" >> performance_test_results.txt
echo "" >> performance_test_results.txt

echo "--- Query Plan: GetCallStatsByCompanyRange ---" >> performance_test_results.txt
mysql -u root -proot omnicall -e "EXPLAIN SELECT COUNT(*) as total_calls, SUM(CASE WHEN answered = 1 THEN 1 ELSE 0 END) as answered_calls, SUM(CASE WHEN answered = 0 THEN 1 ELSE 0 END) as missed_calls, AVG(CASE WHEN duration > 0 THEN duration ELSE NULL END) as avg_duration FROM transcriptions WHERE company_id = 1 AND created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY) AND created_at < NOW()" >> performance_test_results.txt
echo "" >> performance_test_results.txt

# EXPLAIN task stats queries
echo "--- Query Plan: GetTaskStatsByCompanyToday ---" >> performance_test_results.txt
mysql -u root -proot omnicall -e "EXPLAIN SELECT COUNT(*) as total_tasks, SUM(CASE WHEN status = 'completed' THEN 1 ELSE 0 END) as completed_tasks, SUM(CASE WHEN status = 'pending' THEN 1 ELSE 0 END) as pending_tasks, SUM(CASE WHEN status = 'follow-up' THEN 1 ELSE 0 END) as follow_up_tasks, SUM(CASE WHEN due_date < NOW() AND status != 'completed' THEN 1 ELSE 0 END) as overdue_tasks FROM tasks WHERE company_id = 1 AND DATE(created_at) = CURDATE()" >> performance_test_results.txt
echo "" >> performance_test_results.txt

echo "--- Query Plan: GetTaskStatsByCompanyRange ---" >> performance_test_results.txt
mysql -u root -proot omnicall -e "EXPLAIN SELECT COUNT(*) as total_tasks, SUM(CASE WHEN status = 'completed' THEN 1 ELSE 0 END) as completed_tasks, SUM(CASE WHEN status = 'pending' THEN 1 ELSE 0 END) as pending_tasks, SUM(CASE WHEN status = 'follow-up' THEN 1 ELSE 0 END) as follow_up_tasks, SUM(CASE WHEN due_date < NOW() AND status != 'completed' THEN 1 ELSE 0 END) as overdue_tasks FROM tasks WHERE company_id = 1 AND created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY) AND created_at < NOW()" >> performance_test_results.txt
echo "" >> performance_test_results.txt

echo "Test script complete. Check performance_test_results.txt for EXPLAIN output."
EOF

chmod +x run_performance_tests.sh
./run_performance_tests.sh
```
