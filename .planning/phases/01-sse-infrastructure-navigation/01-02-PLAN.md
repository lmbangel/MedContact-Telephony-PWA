2. On load, check authentication (same pattern as dashboard-home.js):
```javascript
authService.init().then(() => {
  if (!authService.isAuthenticated()) {
    window.location.href = '/app/login.html';
    return;
  }

  // Check role authorization
  const user = authService.getCurrentUser();
  const allowedRoles = ['admin', 'manager', 'supervisor', 'support'];
  if (!allowedRoles.includes(user.role)) {
    console.log('Unauthorized role for stats page:', user.role);
    window.location.href = '/app/dashboard-home.html';
    return;
  }

  initializeStatsPage();
});
```

3. Initialize SSE connection with status updates:
```javascript
function initializeStatsPage() {
  // Set up SSE listeners
  sseService.setListeners({
    onMessage: handleSSEMessage,
    onStatusChange: updateConnectionStatus,
    onError: handleSSEError
  });

  // Connect to SSE stream
  sseService.connect(`${API_URL}/api/stats/stream`);

  // Setup cleanup on page unload
  window.addEventListener('beforeunload', cleanup);
}

function cleanup() {
  sseService.disconnect();
}
```

4. Update UI based on connection status:
```javascript
function updateConnectionStatus(status) {
  const indicator = document.getElementById('sse-status-indicator');
  const text = document.getElementById('sse-status-text');
  const debugStatus = document.getElementById('debug-status');
  const debugAttempts = document.getElementById('debug-attempts');

  // Update indicator color
  indicator.classList.remove('bg-gray-400', 'bg-green-500', 'bg-yellow-500', 'bg-red-500');

  switch(status) {
    case 'connected':
      indicator.classList.add('bg-green-500');
      text.textContent = 'Connected';
      break;
    case 'connecting':
      indicator.classList.add('bg-yellow-500');
      text.textContent = 'Connecting...';
      break;
    case 'disconnected':
      indicator.classList.add('bg-red-500');
      text.textContent = 'Disconnected';
      break;
  }

  if (debugStatus) debugStatus.textContent = status;
  if (debugAttempts) debugAttempts.textContent = sseService.reconnectAttempts;
}
```

5. Handle SSE messages (heartbeat for now):
```javascript
function handleSSEMessage(data) {
  console.log('SSE message:', data);

  if (data.type === 'heartbeat') {
    const debugHeartbeat = document.getElementById('debug-heartbeat');
    if (debugHeartbeat) {
      debugHeartbeat.textContent = new Date(data.timestamp).toLocaleTimeString();
    }
  }
}
```
